let model; let labelMap = new Map(); let lastPositions = new Map(); let hitTestSource = null; let xrRefSpace = null; let currentHitMatrix = null; init(); async function init() { // ===== THREE ===== scene = new THREE.Scene(); camera = new THREE.PerspectiveCamera(70, innerWidth/innerHeight, 0.01, 20); renderer = new THREE.WebGLRenderer({ antialias:true, alpha:true }); renderer.setSize(innerWidth, innerHeight); renderer.xr.enabled = true; document.body.appendChild(renderer.domElement); // ===== AR BUTTON ===== document.body.appendChild( ARButton.createButton(renderer, { requiredFeatures: ['hit-test'] }) ); scene.add(new THREE.HemisphereLight(0xffffff, 0xbbbbff, 1)); // ===== XR SESSION ===== renderer.xr.addEventListener("sessionstart", async () => { const session = renderer.xr.getSession(); xrRefSpace = await session.requestReferenceSpace("viewer"); hitTestSource = await session.requestHitTestSource({ space: xrRefSpace }); }); // ===== AI MODEL ===== model = await cocoSsd.load(); overlay("AI připravena"); renderer.setAnimationLoop(render); } let lastDetect = 0; async function render(time, frame) { // ===== HIT TEST ===== if (frame && hitTestSource) { const referenceSpace = renderer.xr.getReferenceSpace(); const hits = frame.getHitTestResults(hitTestSource); if (hits.length > 0) { const hit = hits[0]; const pose = hit.getPose(referenceSpace); currentHitMatrix = pose.transform.matrix; } } // ===== AI LOOP ===== if (model && time - lastDetect > 300) { lastDetect = time; // fallback: použijeme XR framebuffer jako "fake video" const predictions = await model.detect(renderer.domElement); handlePredictions(predictions); } renderer.render(scene, camera); } function handlePredictions(predictions) { const used = new Set(); predictions.forEach(p => { if (p.score < 0.6) return; const key = p.class; const centerX = (p.bbox[0] + p.bbox[2]/2) / window.innerWidth; const centerY = (p.bbox[1] + p.bbox[3]/2) / window.innerHeight; let pos = placeStable(centerX, centerY); if (!pos) return; pos = smoothPosition(key, pos); let label = labelMap.get(key); if (!label) { label = createLabel(""); scene.add(label); labelMap.set(key, label); } updateLabel(label, p); label.position.copy(pos); used.add(key); }); labelMap.forEach((label, key) => { if (!used.has(key)) { scene.remove(label); labelMap.delete(key); } }); } // ===== STABLE POSITION ===== function placeStable(x, y) { if (!currentHitMatrix) return null; const base = new THREE.Matrix4().fromArray(currentHitMatrix); const position = new THREE.Vector3().setFromMatrixPosition(base); const offset = projectTo3D(x, y); return position.lerp(offset, 0.3); } // ===== PROJECTION ===== function projectTo3D(x, y) { const vec = new THREE.Vector3( x * 2 - 1, -(y * 2 - 1), 0.5 ); vec.unproject(camera); const dir = vec.sub(camera.position).normalize(); return camera.position.clone().add(dir.multiplyScalar(2)); } // ===== SMOOTH ===== function smoothPosition(key, newPos) { if (!lastPositions.has(key)) { lastPositions.set(key, newPos.clone()); } const prev = lastPositions.get(key); prev.lerp(newPos, 0.2); return prev; } // ===== LABEL ===== function createLabel() { const canvas = document.createElement('canvas'); canvas.width = 256; canvas.height = 128; const texture = new THREE.CanvasTexture(canvas); const sprite = new THREE.Sprite( new THREE.SpriteMaterial({ map: texture }) ); sprite.scale.set(0.5, 0.25, 1); return sprite; } function updateLabel(sprite, prediction) { const canvas = sprite.material.map.image; const ctx = canvas.getContext('2d'); ctx.clearRect(0,0,256,128); ctx.fillStyle = "black"; ctx.fillRect(0,0,256,128); ctx.fillStyle = "white"; ctx.font = "22px Arial"; ctx.textAlign = "center"; ctx.textBaseline = "middle"; ctx.fillText( prediction.class + " " + Math.round(prediction.score * 100) + "%", 128, 64 ); sprite.material.map.needsUpdate = true; } // ===== UI ===== function overlay(t) { document.getElementById('overlay').innerText = t; }
